\subsection*{4}
اشترت امرأة 3 عطور وكانت قيمة العطر الثاني نصف قيمة العطر الأول وقيمة العطر الثالث نصف قيمة العطر الثاني وكان المجمل 2100 ريال ما سعر العطر الأول؟

\begin{itemize}
    \item أ. 1200
    \item ب. 600
    \item ج. 300
    \item د. 1100
\end{itemize}

\textbf{الحل}
\vspace{0.5em} % Adds a small vertical space for readability
\par العطر الأول + العطر الثاني + العطر الثالث = 2100 ريال
\begin{itemize}
    \item بتجربة الخيارات:
    \item نبدأ بـ الخيار (أ): لو العطر الأول قيمته 1200 ريال.
    \item فإن: \\
          \quad سعر العطر الثاني = $\frac{1}{2} \times \text{سعر العطر الأول} = \frac{1}{2} \times 1200 = 600$ ريال \\
          \quad سعر العطر الثالث = $\frac{1}{2} \times \text{سعر العطر الثاني} = \frac{1}{2} \times 600 = 300$ ريال
    \item ويصبح مجموعهم:
    \[ 1200 + 600 + 300 = 2100 \]
    \item إذاً الخيار (أ) هو الصحيح.
\end{itemize}
