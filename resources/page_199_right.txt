\section*{استراتيجية ١}
\subsection*{قاعدة ١: طريقة التجربة لحل التمارين اللفظية}

من أكثر الطرق استخداماً في حل تمارين القدرات:

\begin{itemize}
    \item \textbf{النوع الأول}
    \begin{itemize}
        \item صنع معادلة من معطيات التمرين ثم نجرب الخيارات عليها.
    \end{itemize}
    \item \textbf{النوع الثاني}
    \begin{itemize}
        \item تخمين ذكي من الخيارات عن العدد الذي يحقق التمرين.
    \end{itemize}
\end{itemize}

\vspace{1em} % مسافة فاصلة

\noindent \textbf{١) حقيبة وكتاب قيمتها ٤٨ ريال إذا كان سعر الكتاب نصف سعر الحقيبة أوجد سعر الكتاب}
\begin{itemize}
    \item[أ)] ١٦ ريال
    \item[ب)] ٣٢ ريال
    \item[ج)] ٢٤ ريال
    \item[د)] ٢١ ريال
\end{itemize}

\textbf{الحل}
$\text{حقيبة} + \text{كتاب} = 48 \text{ ريال} \quad (1)$

نجرب الخيارات على المعادلة:
لو الكتاب $16$ ريال تكون الحقيبة بـ $32$ ريال.
نعوض في المعادلة (1): $16 + 32 = 48$
أي أن الحل صحيح (أ)

\textbf{حل أسرع}
$\text{الحقيبة} = 2 \times \text{كتاب}$
قسم $48 \div 3 = 16$ (أو باستخدام الكسر: $\frac{48}{3} = 16$)
يكون الكتاب بـ $16$ ريال.
والحقيبة بـ $32$ ريال.

\vspace{1em} % مسافة فاصلة

\noindent \textbf{٢) إذا كان ثمن قلم وكتاب ٧٢ ريال، فما سعر الكتاب إذا كان يساوي ثلاثة أمثال القلم؟}
\begin{itemize}
    \item[أ)] ٥٤
    \item[ب)] ٥٢
    \item[ج)] ٣٦
    \item[د)] ٤٢
\end{itemize}

\textbf{الحل}
$\text{كتاب} + \text{قلم} = 72 \text{ ريال}$

نجرب الخيارات: لو الكتاب بـ $54$ ريال يكون القلم بـ $18$ ريال.
وبذلك يكون $\text{كتاب} + \text{قلم} = 54 + 18 = 72$
أي أن الحل (أ) هو الحل الصحيح.

\vspace{1em} % مسافة فاصلة

\noindent \textbf{٣) إذا كان ثلاثة آلات حاسبة و قلم = ٩٠ ريال و ٩ آلات حاسبة و قلمين = ٢٤٠ ريال كم سعر الآلة الحاسبة؟}
\begin{itemize}
    \item[أ)] ٢٠
    \item[ب)] ٣٠
    \item[ج)] ٤٠
    \item[د)] ٥٠
\end{itemize}

\textbf{الحل}
$3 \times \text{آلة حاسبة} + \text{قلم} = 90$
$9 \times \text{آلة حاسبة} + 2 \times \text{قلم} = 240$

نجرب الخيارات ونخمن قيمة الآلة الحاسبة:
نجد أن قيمة الآلة $= 20$ والقلم $= 30$.
(للتحقق: $3 \times 20 + 30 = 60 + 30 = 90$)
(للتحقق: $9 \times 20 + 2 \times 30 = 180 + 60 = 240$)
وهي قيم تحقق معادلات التمرين (أ).
