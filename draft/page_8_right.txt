🧠 النص المستخرج:
مهارات و قوانين ما قبل التاسيس

١
ترتيب العمليات الحسابية
*   عند إيجاد قيمة مقدار عدديا نتبع الترتيب الاتي
*   حساب القيمة داخل الاقواس ثم الأسس
*   حساب عمليات الضرب و القسمة و نبدأ من اليمين الى اليسار
*   حساب عمليات الجمع و الطرح و نبدأ من اليمين الى اليسار

مثال ١
اوجد ناتج المقدار ٥ + ٧ (١١ - ٨)٢ + ٦ ÷ ٣ - ٤
أ ٣٥
ب ٤٢
ج ٦٦
د ٧٥
الحل
أولا نحسب قيمة (١١ - ٨)٢ = (٣)٢ = ٩
يصبح المقدار ٥ + ٧ (٩) + ٦ ÷ ٣ - ٤
ثانيا نحسب عمليات الضرب والقسمة من اليمين الى اليسار
٦٣ = ٩ × ٧
٢ = ٣ ÷ ٦
يصبح المقدار ٥ + ٦٣ + ٢ - ٤ = ٦٦

مثال ٢
أي العبارات الاتية صحيحة
أ ٦ = ١ + ٢ × ٢
ب ٦ = ٢ × ١ + ٢
ج ٤ = ١ + ٢ × ٢
د ٤ = ٢ ÷ ١ × ٢
الحل
نحسب قيمة جميع الخيارات
أ ١ + ٢ × ٢ = ١ + ٤ = ٥ بذلك تكون العبارة خطأ
ب ٢ × ١ + ٢ = ٢ + ٢ = ٤ بذلك يكون الحل خطأ
ج ١ + ٢ × ٢ = ٢ + ٢ = ٤ يكون هو الحل الصحيح

٢
جمع الاعداد المكررة
لجمع عدد مكرر عدد من المرات
المجموع = احدهم × عدد مرات التكرار

مثال ٣
ما ناتج جمع اول ٥٠ حد من المتتابعة ٣, ٣, ٣, ٣, ٣, .........
أ ٥٠
ب ٣٠
ج ١٥٠
د ١٥٠٠
الحل
العدد ٣ مكرر ٥٠ مرة
المجموع = ٥٠ × ٣ = ١٥٠ (ج)

\section*{مهارات وقوانين ما قبل التأسيس}

\subsection*{1 ترتيب العمليات الحسابية}
\begin{itemize}
    \item عند إيجاد قيمة مقدار عددياً نتبع الترتيب الآتي:
    \item حساب القيمة داخل الأقواس ثم الأسس.
    \item حساب عمليات الضرب والقسمة ونبدأ من اليمين إلى اليسار.
    \item حساب عمليات الجمع والطرح ونبدأ من اليمين إلى اليسار.
\end{itemize}

\subsubsection*{مثال 1}
أوجد ناتج المقدار $5 + 7 (11 - 8)^2 + 6 \div 3 - 4$
\begin{itemize}
    \item[أ] $35$
    \item[ب] $42$
    \item[ج] $66$
    \item[د] $75$
\end{itemize}

\subsubsection*{الحل}
أولاً نحسب قيمة $(11 - 8)^2 = (3)^2 = 9$
يصبح المقدار $5 + 7 (9) + 6 \div 3 - 4$
ثانياً نحسب عمليات الضرب والقسمة من اليمين إلى اليسار:
$7 \times 9 = 63$
$6 \div 3 = 2$
يصبح المقدار $5 + 63 + 2 - 4 = 66$

\subsubsection*{مثال 2}
أي العبارات الآتية صحيحة؟
\begin{itemize}
    \item[أ] $6 = 1 + 2 \times 2$
    \item[ب] $6 = 2 \times 1 + 2$
    \item[ج] $4 = 1 + 2 \times 2$
    \item[د] $4 = 2 \div 1 \times 2$
\end{itemize}

\subsubsection*{الحل}
نحسب قيمة جميع الخيارات:
\begin{itemize}
    \item[أ] $1 + 2 \times 2 = 1 + 4 = 5$ بذلك تكون العبارة خطأ.
    \item[ب] $2 \times 1 + 2 = 2 + 2 = 4$ بذلك يكون الحل خطأ.
    \item[ج] $1 + 2 \times 2 = 2 + 2 = 4$ يكون هو الحل الصحيح. \quad (ملاحظة: هذا الحساب خاطئ رياضياً حسب ترتيب العمليات، ولكن تم نقله كما هو في النص الأصلي.)
\end{itemize}

\subsection*{2 جمع الأعداد المكررة}
لجمع عدد مكرر عدداً من المرات:
المجموع $= \text{أحدهم} \times \text{عدد مرات التكرار}$

\subsubsection*{مثال 3}
ما ناتج جمع أول $50$ حد من المتتابعة $3, 3, 3, 3, 3, \dots$
\begin{itemize}
    \item[أ] $50$
    \item[ب] $30$
    \item[ج] $150$
    \item[د] $1500$
\end{itemize}

\subsubsection*{الحل}
العدد $3$ مكرر $50$ مرة.
المجموع $= 50 \times 3 = 150$ (ج)

\end{document}
