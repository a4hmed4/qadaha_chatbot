٢٧ زوجي أم فردي
*   لتحديد المقدار الزوجي
    *   نعوض بعدد زوجي مكان المجهول بالمقدار
*   لتحديد المقدار الفردي
    *   نعوض بعدد فردي مكان المجهول بالمقدار

مثال ٨٢
إذا كان ن عدد زوجي فأي مما يلي يكون فردي
أ ن² + ٤
ب ن + ٤
جـ ٢ن² + ٣
د ٣ن - ٢

الحل
حيث ن عدد زوجي نعوض عن ن بـ ٢ في جميع الخيارات
أ ٢² + ٤ = ٤ + ٤ = ٨
ب ٢ + ٤ = ٦
جـ ٢ × ٢² + ٣ = ٢ × ٤ + ٣ = ٨ + ٣ = ١١
د ٣ × ٢ - ٢ = ٦ - ٢ = ٤
العدد الفردي هو جـ ٢ن² + ٣

مثال ٨٣
إذا كان ن عدد فردي، فأي الآتي زوجي ؟
أ ٣ (ن - ١)
ب ٣ن
جـ ن + ٢
د ٣ (ن + ٢)

الحل
حيث ن عدد فردي نعوض عن ن بـ ٣ في جميع الخيارات
أ ٣ (٣ - ١) = ٣ × ٢ = ٦
ب ٣ × ٣ = ٩
جـ ٣ + ٢ = ٥
د ٣ (٣ + ٢) = ٣ × ٥ = ١٥
الوحيد الزوجي هو أ ٣ (ن - ١)

مثال ٨٤
(م ، ن ، ل) أعداد فردية، أي من التالي ليس زوجياً ؟
أ ن + م + ل - ١١
ب ن + ١
جـ ن + م + ١
د ن - م

الحل
نعوض عن م , ن , ل بأعداد فردية مثلا ١ , ٣ , ٥ بالترتيب
أ ١ + ٣ + ٥ - ١١ = ٩ - ١١ = -٢
ب ٣ + ١ = ٤
جـ ١ + ٣ + ١ = ٥
د ٣ - ١ = ٢
الوحيد الفردي هو جـ ن + م + ١

```latex
التأسيس
فيديو ١٤

٢٧ زوجي أم فردي

\begin{itemize}
    \item لتحديد المقدار الزوجي
    \begin{itemize}
        \item نعوض بعدد زوجي مكان المجهول بالمقدار.
    \end{itemize}
    \item لتحديد المقدار الفردي
    \begin{itemize}
        \item نعوض بعدد فردي مكان المجهول بالمقدار.
    \end{itemize}
\end{itemize}

\textbf{مثال ٨٢}
إذا كان $ن$ عدد زوجي فأي مما يلي يكون فردي؟
\begin{enumerate}[أ)]
    \item $ن^2 + 4$
    \item $ن + 4$
    \item $2ن^2 + 3$
    \item $3ن - 2$
\end{enumerate}

\textbf{الحل}
حيث $ن$ عدد زوجي، نعوض عن $ن$ بـ $2$ في جميع الخيارات:
\begin{enumerate}[أ)]
    \item $2^2 + 4 = 4 + 4 = 8$
    \item $2 + 4 = 6$
    \item $2 \times 2^2 + 3 = 2 \times 4 + 3 = 8 + 3 = 11$
    \item $3 \times 2 - 2 = 6 - 2 = 4$
\end{enumerate}
العدد الفردي هو جـ) $2ن^2 + 3$.

\textbf{مثال ٨٣}
إذا كان $ن$ عدد فردي، فأي الآتي زوجي؟
\begin{enumerate}[أ)]
    \item $3 (ن - 1)$
    \item $3ن$
    \item $ن + 2$
    \item $3 (ن + 2)$
\end{enumerate}

\textbf{الحل}
حيث $ن$ عدد فردي، نعوض عن $ن$ بـ $3$ في جميع الخيارات:
\begin{enumerate}[أ)]
    \item $3 (3 - 1) = 3 \times 2 = 6$
    \item $3 \times 3 = 9$
    \item $3 + 2 = 5$
    \item $3 (3 + 2) = 3 \times 5 = 15$
\end{enumerate}
الوحيد الزوجي هو أ) $3 (ن - 1)$.

\textbf{مثال ٨٤}
($م$، $ن$، $ل$) أعداد فردية، أي من التالي ليس زوجياً؟
\begin{enumerate}[أ)]
    \item $ن + م + ل - 11$
    \item $ن + 1$
    \item $ن + م + 1$
    \item $ن - م$
\end{enumerate}

\textbf{الحل}
نعوض عن $م$، $ن$، $ل$ بأعداد فردية مثلاً $1$، $3$، $5$ بالترتيب:
\begin{enumerate}[أ)]
    \item $1 + 3 + 5 - 11 = 9 - 11 = -2$
    \item $3 + 1 = 4$
    \item $1 + 3 + 1 = 5$
    \item $3 - 1 = 2$
\end{enumerate}
الوحيد الفردي هو جـ) $ن + م + 1$.

١٦
استخدم تنسيقات $\frac{}{}$، $\sqrt{}$، $^2$، $=$، $<$، $>$، إلخ.
```
