\paragraph*{الطريقة الأفقية}
لابد من جعل العلامات العشرية متساوية عن طريق وضع أصفار على يمين العدد

\textbf{مثال}\\
$2.3 + 4.5$\\
العلامات موحدة فسوف نجمع ونضع العلامة كما هي\\
ليصبح الناتج $6.8$

\textbf{مثال}\\
$2.45 + 1.2$\\
لابد أن نضع $0$ بعد العدد $2$ كي تتساوى العلامات\\
$2.45 + 1.20$\\
ونجمع كل رقم مع المقابل له ليصبح الناتج هو $3.65$

\paragraph*{الطريقة الرأسية}
وهي تعتمد على وضع الأعداد فوق بعضها البعض\\
بشرط وضع العلامات العشرية تحت بعض

\vspace{1em} % Add some vertical space

\textbf{1}\\
ما قيمة المقدار $0.007 + 0.07 + 0.7 + 7$\\
\begin{enumerate}[label={\alph*}, itemsep=0pt]
    \item $7.774$
    \item $7.784$
    \item $7.777$
    \item $7.874$
\end{enumerate}

\textbf{الحل}\\
نجمع رأسي\\
ليصبح الناتج $7.784$
\begin{equation*}
\begin{array}{r @{.} l} % Right align before decimal, left align after
     7 & 000 \\
     0 & 700 \\
     0 & 070 \\
+    0 & 007 \\
\hline
     7 & 777 \\
\end{array}
\end{equation*}

\textbf{2}\\
ما قيمة $1.8 + 0.8 + 0.08 + 0.008$\\
\begin{enumerate}[label={\alph*}, itemsep=0pt]
    \item $1.888$
    \item $2.688$
    \item $1.88$
    \item $2.768$
\end{enumerate}

\textbf{الحل}\\
نجمع رأسي\\
ليصبح الناتج هو $2.688$
\begin{equation*}
\begin{array}{r @{.} l}
      1 & 800 \\
      0 & 800 \\
      0 & 080 \\
+     0 & 008 \\
\hline
      2 & 688 \\
\end{array}
\end{equation*}

\textbf{3}\\
ما قيمة $1.1 + 1.01 + 1.11 + 1.011$\\
\begin{enumerate}[label={\alph*}, itemsep=0pt]
    \item $4.231$
    \item $4.444$
    \item $3.421$
    \item $4.111$
\end{enumerate}

\textbf{الحل}\\
نجمع رأسي\\
ليصبح الناتج هو $4.231$
\begin{equation*}
\begin{array}{r @{.} l}
      1 & 100 \\
      1 & 010 \\
      1 & 110 \\
+     1 & 011 \\
\hline
      4 & 231 \\
\end{array}
\end{equation*}
