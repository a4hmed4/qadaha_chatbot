مهارات و قوانين ما قبل الـ
II فك القوس التربيع
(س + ص)² = س² + ٢ س ص + ص²
(س - ص)² = س² - ٢ س ص + ص²

مثال ٣٥
قارن بين
القيمة الأولى (س + ص)²
القيمة الثانية س² + ٢ س ص + ص²
أ القيمة الأولى أكبر
ب القيمة الثانية أكبر
ج القيمتان متساويتان
د المعلومات غير كافية
الحل
حيث ان ( س + ص )² = س² + ٢ س ص + ص²
فإن القيمتان متساويتان (ج)

مثال ٣٦
قارن بين
القيمة الأولى (س + ص)²
القيمة الثانية س² + ص²
أ القيمة الأولى أكبر
ب القيمة الثانية أكبر
ج القيمتان متساويتان
د المعلومات غير كافية
الحل
القيمة الأولى (س + ص)² = س² + ٢ س ص + ص²
القيمة الثانية س² + ص²
وحيث اننا لا نستطيع تحديد قيمة س, ص فلا نستطيع معرفة
ايهما اكبر لذلك تكون الإجابة (د)

مثال ٣٧
تبسيط المقدار (أ + ب)² - (أ - ب)²
أ) ٢ ب²
ب) ٤ أ ب
ج) أ ب
د) ٤ أ ب
الحل
بفك الاقواس
(أ² + ٢ أ ب + ب²) - (أ² - ٢ أ ب + ب²) =
أ² + ٢ أ ب + ب² - أ² + ٢ أ ب - ب² = ٤ أ ب (د)

مثال ٣٨
إذا كان أ² + ب² = ٤، أ + ب = ٨ اوجد أ ب
أ) ٦٤
ب) ٦٠
ج) ٣٠
د) ١٢
الحل
(أ + ب)² = أ² + ٢ أ ب + ب²
٨² = ٤ + ٢ أ ب
٦٤ = ٤ + ٢ أ ب
٦٤ - ٤ = ٢ أ ب
٦٠ = ٢ أ ب
٣٠ = أ ب
أي أن أ ب = ٣٠.

\textbf{مهارات و قوانين ما قبل الـ}

\textbf{II فك القوس التربيع}
\[ (\text{س} + \text{ص})^2 = \text{س}^2 + 2 \text{س} \text{ص} + \text{ص}^2 \]
\[ (\text{س} - \text{ص})^2 = \text{س}^2 - 2 \text{س} \text{ص} + \text{ص}^2 \]

\textbf{مثال ٣٥}
\textbf{قارن بين}
\textbf{القيمة الأولى} $(\text{س} + \text{ص})^2$
\textbf{القيمة الثانية} $\text{س}^2 + 2 \text{س} \text{ص} + \text{ص}^2$
أ) القيمة الأولى أكبر
ب) القيمة الثانية أكبر
ج) القيمتان متساويتان
د) المعلومات غير كافية

\textbf{الحل}
حيث أن $(\text{س} + \text{ص})^2 = \text{س}^2 + 2 \text{س} \text{ص} + \text{ص}^2$
فإن القيمتان متساويتان (ج)

\textbf{مثال ٣٦}
\textbf{قارن بين}
\textbf{القيمة الأولى} $(\text{س} + \text{ص})^2$
\textbf{القيمة الثانية} $\text{س}^2 + \text{ص}^2$
أ) القيمة الأولى أكبر
ب) القيمة الثانية أكبر
ج) القيمتان متساويتان
د) المعلومات غير كافية

\textbf{الحل}
القيمة الأولى $(\text{س} + \text{ص})^2 = \text{س}^2 + 2 \text{س} \text{ص} + \text{ص}^2$
القيمة الثانية $\text{س}^2 + \text{ص}^2$
وحيث أننا لا نستطيع تحديد قيمة \text{س}, \text{ص} فلا نستطيع معرفة أيهما أكبر لذلك تكون الإجابة (د).

\textbf{مثال ٣٧}
\textbf{تبسيط المقدار} $(\text{أ} + \text{ب})^2 - (\text{أ} - \text{ب})^2$
أ) $2 \text{ب}^2$
ب) $4 \text{أ} \text{ب}$
ج) $\text{أ} \text{ب}$
د) $4 \text{أ} \text{ب}$

\textbf{الحل}
بفك الأقواس:
\begin{align*}
(\text{أ}^2 + 2 \text{أ} \text{ب} + \text{ب}^2) - (\text{أ}^2 - 2 \text{أ} \text{ب} + \text{ب}^2) &= \\
\text{أ}^2 + 2 \text{أ} \text{ب} + \text{ب}^2 - \text{أ}^2 + 2 \text{أ} \text{ب} - \text{ب}^2 &= 4 \text{أ} \text{ب} \quad (\text{د})
\end{align*}

\textbf{مثال ٣٨}
\textbf{إذا كان} $\text{أ}^2 + \text{ب}^2 = 4$\textbf{،} $\text{أ} + \text{ب} = 8$\textbf{ أوجد} $\text{أ} \text{ب}$
أ) $64$
ب) $60$
ج) $30$
د) $12$

\textbf{الحل}
$(\text{أ} + \text{ب})^2 = \text{أ}^2 + 2 \text{أ} \text{ب} + \text{ب}^2$
$8^2 = 4 + 2 \text{أ} \text{ب}$
$64 = 4 + 2 \text{أ} \text{ب}$
$64 - 4 = 2 \text{أ} \text{ب}$
$60 = 2 \text{أ} \text{ب}$
$30 = \text{أ} \text{ب}$
أي أن $\text{أ} \text{ب} = 30$.
