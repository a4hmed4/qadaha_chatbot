\begin{enumerate}[itemsep=1.5em]

\item عند كتابة برقية فإن أول $15$ كلمة بريالين وبعدها تصبح الكلمة بـ $12.5$ هللة، كم كلمة في برقية ثمنها $5$ ريال ؟
\begin{enumerate}[a)]
    \item $36$ كلمة
    \item $40$ كلمة
    \item $39$ كلمة
    \item $30$ كلمة
\end{enumerate}
\textbf{الحل:} \\
تكلفة أول $15$ كلمة = $2$ ريال = $2 \times 100 = 200$ هللة. \\
المبلغ المتبقي = $5$ ريال - $2$ ريال = $3$ ريال = $3 \times 100 = 300$ هللة. \\
عدد الكلمات الإضافية = $\frac{300}{12.5} = 24$ كلمة. \\
إجمالي عدد الكلمات = $15 + 24 = 39$ كلمة.

\item إذا كان تقاضى محمود $1125$ ريال وكان يأخذ $75$ ريال في الساعة الواحدة فكم ساعة عمل ؟
\begin{enumerate}[a)]
    \item $13$
    \item $14$
    \item $15$
    \item $16$
\end{enumerate}
\textbf{الحل:} \\
عدد الساعات = $\frac{\text{إجمالي المبلغ}}{\text{الأجر في الساعة الواحدة}} = \frac{1125}{75} = 15$ ساعة.

\item تنتج شركة $4800$ صحيفة في $12$ يوم وشركة أخرى $3600$ صحيفة في تسعة أيام، قارن بين:
\begin{itemize}
    \item \textbf{القيمة الأولى:} ما تنتجه الأولى في اليوم
    \item \textbf{القيمة الثانية:} ما تنتجه الثانية في اليوم
\end{itemize}
\begin{enumerate}[a)]
    \item القيمة الأولى أكبر
    \item القيمة الثانية أكبر
    \item القيمتان متساويتان
    \item المعطيات غير كافية
\end{enumerate}
\textbf{الحل:} \\
ما تنتجه الشركة الأولى في اليوم = $\frac{4800}{12} = 400$ صحيفة/يوم. \\
ما تنتجه الشركة الثانية في اليوم = $\frac{3600}{9} = 400$ صحيفة/يوم. \\
إذن، القيمتان متساويتان.

\item قارن بين:
\begin{itemize}
    \item \textbf{القيمة الأولى:} عدد ينقص $2$ عن $-8$
    \item \textbf{القيمة الثانية:} عدد يزيد $3$ عن $-6$
\end{itemize}
\begin{enumerate}[a)]
    \item القيمة الأولى أكبر
    \item القيمة الثانية أكبر
    \item القيمتان متساويتان
    \item المعطيات غير كافية
\end{enumerate}
\textbf{الحل:} \\
القيمة الأولى = $-8 - 2 = -10$. \\
القيمة الثانية = $-6 + 3 = -3$. \\
بما أن $-10 < -3$ ، إذن القيمة الثانية أكبر.

\item قرأ رجل عبارة في صحيفة وهي: مئة مليون برميل ومئة ألف جالون، أي الآتي يمثل عدد البراميل والجالونات ؟
\begin{enumerate}[a)]
    \item $10^4$ برميل و $10^5$ جالون
    \item $10^8$ برميل و $10^5$ جالون
    \item $10^6$ برميل و $10^5$ جالون
    \item $10^5$ برميل و $10^6$ جالون
\end{enumerate}
\textbf{الحل:} \\
مئة مليون برميل = $100,000,000$ برميل = $10^8$ برميل. \\
مئة ألف جالون = $100,000$ جالون = $10^5$ جالون. \\
إذن، الخيار الصحيح هو $10^8$ برميل و $10^5$ جالون.

\item ما قيمة س في المعادلة: $2\text{س} - 1 = 4\text{س} - 3 (2 - \text{س})$ ؟
\begin{enumerate}[a)]
    \item $-1$
    \item $1$
    \item $2$
    \item $3$
\end{enumerate}
\textbf{الحل:} \\
$2\text{س} - 1 = 4\text{س} - 3 (2 - \text{س})$ \\
$2\text{س} - 1 = 4\text{س} - 6 + 3\text{س}$ \\
$2\text{س} - 1 = 7\text{س} - 6$ \\
نقل حدود $\text{س}$ لطرف والأرقام للطرف الآخر: \\
$-1 + 6 = 7\text{س} - 2\text{س}$ \\
$5 = 5\text{س}$ \\
$\text{س} = \frac{5}{5}$ \\
$\text{س} = 1$.

\item ما هو العدد الذي الفرق بينه وبين جذره التربيعي يساوي $72$؟
\begin{enumerate}[a)]
    \item $6$
    \item $81$
    \item $64$
    \item $8$
\end{enumerate}
\textbf{الحل:} \\
ليكن العدد هو $\text{س}$. \\
المعادلة هي: $\text{س} - \sqrt{\text{س}} = 72$. \\
لحل هذه المعادلة، نفرض أن $\text{ص} = \sqrt{\text{س}}$. إذن، $\text{س} = \text{ص}^2$. \\
نعوض في المعادلة: \\
$\text{ص}^2 - \text{ص} = 72$ \\
$\text{ص}^2 - \text{ص} - 72 = 0$ \\
نحل المعادلة التربيعية بالتحليل: \\
$(\text{ص} - 9)(\text{ص} + 8) = 0$ \\
إذن، إما $\text{ص} - 9 = 0 \Rightarrow \text{ص} = 9$ أو $\text{ص} + 8 = 0 \Rightarrow \text{ص} = -8$. \\
بما أن $\text{ص} = \sqrt{\text{س}}$، يجب أن تكون قيمة $\text{ص}$ غير سالبة، لذا نرفض $\text{ص} = -8$. \\
إذن، $\text{ص} = 9$. \\
لإيجاد قيمة $\text{س}$، نربع الطرفين: \\
$\text{س} = \text{ص}^2 = 9^2 = 81$.

\item أي مما يلي لا يصلح أن تكون أضلاعاً لمثلث ؟
\begin{enumerate}[a)]
    \item $3, 4, 5$
    \item $5, 5, 5$
    \item $1, 3, 5$
    \item $2, 6, 7$
\end{enumerate}
\textbf{الحل:} \\
شرط تكون أضلاع مثلث هو أن مجموع طولي أي ضلعين يجب أن يكون أكبر من طول الضلع الثالث. \\
\begin{enumerate}[a)]
    \item $3, 4, 5$: $3+4=7 > 5$ (صحيح)، $3+5=8 > 4$ (صحيح)، $4+5=9 > 3$ (صحيح). \textbf{يصلح}.
    \item $5, 5, 5$: $5+5=10 > 5$ (صحيح). \textbf{يصلح}.
    \item $1, 3, 5$: $1+3=4$. هل $4 > 5$؟ (خطأ). \textbf{لا يصلح}.
    \item $2, 6, 7$: $2+6=8 > 7$ (صحيح)، $2+7=9 > 6$ (صحيح)، $6+7=13 > 2$ (صحيح). \textbf{يصلح}.
\end{enumerate}
إذن، المجموعة التي لا تصلح هي $1, 3, 5$.

\end{enumerate}
