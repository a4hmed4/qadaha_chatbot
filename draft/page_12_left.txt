\subsection*{مثال ٣٠}
\noindent\textbf{قارن بين}
\begin{itemize}
    \item[] \textbf{القيمة الأولى:} أكبر عامل أولي للعدد $١٣٠$
    \item[] \textbf{القيمة الثانية:} $١٢$
\end{itemize}
\begin{enumerate}[\text{أ}]
    \item القيمة الأولى أكبر
    \item القيمة الثانية أكبر
    \item القيمتان متساويتان
    \item المعلومات غير كافية
\end{enumerate}

\noindent\textbf{الحل}
العوامل الأولية للعدد $١٣٠$: نحلل الـ $١٣٠ = ٢ \times ٥ \times ١٣$ \\
أكبر عامل أولي هو $١٣$ \\
لذلك تكون الإجابة (\text{أ}) هي الحل الصحيح.
