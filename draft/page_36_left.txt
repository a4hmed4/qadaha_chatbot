**٤٤.**
\begin{enumerate}
    \item \text{مستطيل طول ضلعه ١١ سم فما أقرب قيمه له بالسم}
    \begin{itemize}
        \item[\text{أ}] $10.6$
        \item[\text{ب}] $11.5$
        \item[\text{ج}] $11.8$
        \item[\text{د}] $10.4$
    \end{itemize}
    \text{الحل:}
    \par \text{عند تقريب الخيارات نجد ( أ ) هو الحل الصحيح}
    \par $10.6 \approx 11 \text{ (أ)}$
\end{enumerate}

**٤٥.**
\begin{enumerate}
    \item \text{أقرب عدد لـ } $6.7$ \text{ هو}
    \begin{itemize}
        \item[\text{أ}] $6.5$
        \item[\text{ب}] $6.69$
        \item[\text{ج}] $6.72$
        \item[\text{د}] $6.77$
    \end{itemize}
    \text{الحل:}
    \par \text{أقرب عدد هو } $6.69 \text{ (ب)}$
\end{enumerate}

**٤٦.**
\begin{enumerate}
    \item \text{قارن بين}
    \par \text{القيمة الأولى } $1.40$
    \par \text{القيمة الثانية } $1.25 \times 1.25$
    \text{الحل:}
    \par \text{القيمة الثانية } $1.25 \times 1.25$
    \par \text{نقرب لجعل الحسابات أسهل}
    \par \text{لتصبح } $1.2 \times 1.2 = 1.44$
    \par \text{وبذلك يتضح أن القيمة الثانية أكبر ( ب )}
\end{enumerate}

**٤٧.**
\begin{enumerate}
    \item \text{قارن بين}
    \par \text{القيمة الأولى } $5$
    \par \text{القيمة الثانية } $\frac{0.9}{1.9}$
    \text{الحل:}
    \par \text{نقرب المقدار } $\frac{0.9}{1.9} = \frac{6}{2} = 3$
    \par \text{وب}
\end{enumerate}
