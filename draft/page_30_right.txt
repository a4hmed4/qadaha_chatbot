```latex
% يجب تضمين الحزم المناسبة للغة العربية
% على سبيل المثال، باستخدام polyglossia (الأفضل):
% \usepackage{polyglossia}
% \setmainlanguage{arabic}
% \newfontfamily\arabicfont[Script=Arabic]{Amiri} % أو أي خط عربي مثبت

% أو باستخدام babel:
% \usepackage[utf8]{inputenc}
% \usepackage[arabic]{babel}
% \preto\section{\R{}} % لجعل عناوين الأقسام من اليمين لليسار
% \preto\subsection{\R{}} % لجعل عناوين الأقسام الفرعية من اليمين لليسار

\documentclass{article} % أو book/report حسب الحاجة
\usepackage{amsmath} % مفيد للرياضيات بشكل عام
\usepackage{ragged2e} % لمزيد من التحكم في المحاذاة

% إعدادات اللغة العربية (مثال باستخدام babel)
\usepackage[utf8]{inputenc}
\usepackage[arabic]{babel}
% إعدادات تجريبية لجعل الأقسام من اليمين لليسار مع babel
\makeatletter
\def\section{%
  \@startsection{section}{1}{\z@}%
    {-3.5ex \@plus -1ex \@minus -.2ex}%
    {2.3ex \@plus .2ex}%
    {\normalfont\Large\bfseries\R{}}}
\def\subsection{%
  \@startsection{subsection}{2}{\z@}%
    {-3.25ex\@plus -1ex\@minus-.2ex}%
    {1.5ex \@plus .2ex}%
    {\normalfont\large\bfseries\R{}}}
\makeatother

\begin{document}
% لتأكيد اتجاه النص من اليمين لليسار للوثيقة بالكامل إذا لم يكن babel/polyglossia كافياً
```
