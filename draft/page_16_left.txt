حسب القاعدة \textcircled{1} الدائرة في الرسم هو الشكل الخارجي

تكون مساحة الدائرة أكبر من مساحة المربع (أ)
\[ \text{مساحة الدائرة} > \text{مساحة المربع (أ)} \]

\textbf{١٩ نقطة المنتصف}

\begin{center}
أ \text{----------} م \text{----------} ب \\
$(\text{س}_1, \text{ص}_1) \quad (\text{س}, \text{ص}) \quad (\text{س}_2, \text{ص}_2)$
\end{center}

إذا كانت م نقطة منتصف القطعة المستقيمة أب
فإن:
\[ \text{م} = \frac{\text{أ}+\text{ب}}{2} \]
\[ \text{س} = \frac{\text{س}_1+\text{س}_2}{2} , \quad \text{ص} = \frac{\text{ص}_1+\text{ص}_2}{2} \]

\textbf{مثال ٥٦}
إذا م نقطة منتصف أب

\begin{center}
أ \text{----------} م \text{----------} ب \\
$(3, \text{ص}) \quad (9, 5) \quad (\text{س}, 6)$
\end{center}

قارن بين
القيمة الأولى قيمة س
القيمة الثانية قيمة ص
أ القيمة الأولى أكبر
ب القيمة الثانية أكبر
ج القيمتان متساويتان
د المعلومات غير كافية

\textbf{الحل}
حيث م نقطة المنتصف فإن:
\[ 5 = \frac{3+\text{س}}{2} \quad \rightarrow \quad 3+\text{س} = 10 \quad \rightarrow \quad \text{س} = 7 \]
\[ 9 = \frac{6+\text{ص}}{2} \quad \rightarrow \quad 6+\text{ص} = 18 \quad \rightarrow \quad \text{ص} = 12 \]
أي أن القيمة الثانية أكبر (ب)

\textbf{مثال ٥٧}
ما إحداثيات النقطة أ في الدائرة م
أ $(8, -2)$
ب $(6, 2)$
ج $(1, 5)$
د $(5, 3)$

\begin{center}
    (3, 1) \\
    م \\
    أ \\
    ب $(-2, 4)$
\end{center}

\textbf{الحل}
\[ \text{م} = \frac{\text{أ}+\text{ب}}{2} \quad \rightarrow \quad 2\text{م} = \text{أ}+\text{ب} \quad \rightarrow \quad \text{أ} = 2\text{م} - \text{ب} \]
\[ \text{أ} = 2(3, 1) - (-2, 4) \]
\[ \text{أ} = (6, 2) - (-2, 4) \]
\[ \text{أ} = (8, -2) \]

\end{Arabic}
