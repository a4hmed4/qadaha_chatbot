\subsection*{٤٦)}
\text{٤٠٠ لتر من الحليب قسمناه على علب ، الأولى ربع الكمية والثانية نصف الكمية ، كم لتر تسع العلبة الثالثة ؟}

\begin{enumerate}
    \item[أ)] $120$
    \item[ب)] $100$
    \item[ج)] $160$
    \item[د)] $200$
\end{enumerate}

\subsubsection*{الحل}
\text{الأولى ربع الكمية} $= 400 \times \frac{1}{4} = 100 \text{ لتر}$ \\
\text{الثانية نصف الكمية} $= 400 \times \frac{1}{2} = 200 \text{ لتر}$ \\
\text{المتبقي} $= 400 - (100 + 200) = 100 \text{ لتر}$

\vspace{0.5cm}

\subsection*{٤٧)}
\text{مزرعة تنتج ٤٠٠ لتر حليب إذا كان ربع الكمية يتم تعبئته في علب نصف لتر ونصف الكمية يتم تعبئته في علب ٢ لتر وباقي الكمية في علب لتر فكم علبة لدينا ؟}

\begin{enumerate}
    \item[أ)] $100 \text{ علبة}$
    \item[ب)] $200 \text{ علبة}$
    \item[ج)] $300 \text{ علبة}$
    \item[د)] $400 \text{ علبة}$
\end{enumerate}

\subsubsection*{الحل}
\text{ربع الكمية هو } $100$ \text{ لتر وعند تعبئتها في علب نصف لتر فنحتاج إلى } $\frac{100}{\frac{1}{2}} = 100 \times 2 = 200 \text{ علبة}$ \\
\text{نصف الكمية وهي } $200$ \text{ لتر ويتم تعبئتها في علب } $2$ \text{ لتر أي نحتاج إلى } $\frac{200}{2} = 100 \text{ علبة}$ \\
\text{باقي الكمية هو } $100$ \text{ لتر ويتم تعبئتها في علب سعتها لتر أي نحتاج إلى } $\frac{100}{1} = 100 \text{ علبة}$ \\
\text{يكون عدد العلب كله هو } $200 + 100 + 100 = 400 \text{ علبة (د)}$

\vspace{0.5cm}

\subsection*{٤٨)}
\text{إذا كان ربع ما مع أحمد هو } $6000$ \text{ ريال فما هو نصف ثلث ما معه}

\begin{enumerate}
    \item[أ)] $3000$
    \item[ب)] $4000$
    \item[ج)] $5000$
    \item[د)] $6000$
\end{enumerate}

\subsubsection*{الحل}
\text{ربع ما مع أحمد هو } $6000$ \text{ فإن ما مع أحمد هو } $6000 \times 4$ \\
\text{نصف ثلث ما لديه هو } $6000 \times 4 \times \frac{1}{3} \times \frac{1}{2} = 4000 \text{ (ب)}$

\vspace{0.5cm}

\subsection*{٤٩)}
\text{اسطوانة مملوءة إلى ربعها ثم أضفنا } $70$ \text{ لتر فأصبحت ثلاث أرباع الاسطوانة مملؤة فكم سعتها}

\begin{enumerate}
    \item[أ)] $140 \text{ لتر}$
    \item[ب)] $160 \text{ لتر}$
    \item[ج)] $120 \text{ لتر}$
    \item[د)] $100 \text{ لتر}$
\end{enumerate}

\subsubsection*{الحل}
\text{الكمية المضافة} $= \frac{3}{4} - \frac{1}{4} = \frac{2}{4} = \frac{1}{2}$ \\
$\frac{1}{2} \text{ الخزان} = 70 \text{ لتر أي ان سعة الخزان كامل} = 140 \text{ لتر (أ)}$
