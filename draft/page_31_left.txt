\subsection*{\centering قاعدة 2: الصحيح والعشري}
عند طرح عدد عشري من آخر صحيح لابد من حذف العلامة ويعوض عنها بأصفار في العدد الصحيح ونطرح ثم نعيد العلامة كما هي في الناتج.

\subsubsection*{\centering مثال}
$1 - 0.999$
\par نحذف العلامة ونعوض عنها بأصفار في العدد الصحيح ليصبح التمرين هو $1000 - 999$ ويكون الناتج $1$.
\par ثم نعيد العلامة كما هي ليصبح الناتج $0.001$.

\item أوجد قيمة $1 - 0.0007$
    \begin{itemize}[label={},leftmargin=1.5em,itemsep=0.5em]
        \item أ $0.9993$
        \item ب $0.993$
        \item ج $0.9003$
        \item د $0.003$
    \end{itemize}
    \textbf{الحل}
    \par نستبدل العلامة العشرية بعد $4$ من الأصفار أمام $1$ ليصبح المقدار $10000 - 7$ وناتجه هو $9993$.
    \par ثم نضع العلامة كما كانت ليصبح الناتج هو $0.9993$ (أ).

\item ما قيمة $8 - 0.8 - 0.08 - 0.008$
    \begin{itemize}[label={},leftmargin=1.5em,itemsep=0.5em]
        \item أ $7.112$
        \item ب $0.7112$
        \item ج $7.012$
        \item د $7112$
    \end{itemize}
    \textbf{الحل}
    % Use align* for multi-line equation alignment
    \begin{align*}
    8 - 0.8 - 0.08 - 0.008 &= 8 - (0.8 + 0.08 + 0.008) \\
                           &= 8 - 0.888
    \end{align*}
    % Vertical sum for 0.8 + 0.08 + 0.008
    \begin{center}
    \begin{tabular}{r}
    $0.8$ \\
    $0.08$ \\
    $+ 0.008$ \\
    \cline{1-1}
    $0.888$ \\
    \end{tabular}
    \end{center}
    \par نستبدل العلامة العشرية بـ $3$ من الأصفار أمام $8$ ليصبح $8000 - 888 = 7112$ ثم نعيد العلامة العشرية مرة أخرى ليصبح الناتج $= 7.112$.

\end{enumerate}
