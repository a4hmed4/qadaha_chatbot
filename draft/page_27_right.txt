\text{تجميعات نماذج المحوسب تستطيع حل}

\bigskip

\text{٥٤ أوجد قيمة أ}
\begin{itemize}
    \item[أ] $0.326$
    \item[ب] $0.300$
    \item[ج] $0.318$
    \item[د] $0.325$
\end{itemize}
\text{ملاحظة: الصياغة الأصلية "أوجد قيمة أ ج ٠,٣١٢" قد تكون غير كاملة أو تحتوي على خطأ مطبعي. تم تقديم الخيارات فقط.}

\bigskip

\text{٥٥ عدد ضرب في نفسه واضيف اليه ثلاثة أمثاله، أي العبارات تمثل ذلك؟}
\begin{itemize}
    \item[أ] $6x^2$
    \item[ب] $3x + 1$
    \item[ج] $x(x + 3)$
    \item[د] $6x + 3$
\end{itemize}

\bigskip

\text{٥٦ مربع بداخله ٣ مربعات بداخل كل مربع ٥ مربعات. كم عدد المربعات؟}
\begin{itemize}
    \item[أ] $15$
    \item[ب] $16$
    \item[ج] $19$
    \item[د] $20$
\end{itemize}

\bigskip

\text{٥٧ أوجد قيمة $1 + 2 + 3 + 4 + 41 + \dots + 1432$}
\begin{itemize}
    \item[أ] $1$
    \item[ب] $1433$
    \item[ج] $1432$
    \item[د] $-1$
\end{itemize}
\text{ملاحظة: وجود الرقم 41 في السلسلة الحسابية يبدو غير متناسق مع النمط. إذا كانت السلسلة $1+2+3+\dots+1432$ فمجموعها هو $1025548$.}

\bigskip

\text{٥٨ أب قرر يعطي ابنه ٨ ريال على كل ٨٠ ريال يجمعها. كم سيعطيه إذا جمع ٨٢٠ ريال؟}
\begin{itemize}
    \item[أ] $80$
    \item[ب] $82$
    \item[ج] $88$
    \item[د] $92$
\end{itemize}
\text{الحل: نسبة ما يعطيه = $\frac{8}{80} = \frac{1}{10}$. المبلغ الذي سيعطيه = $\frac{1}{10} \times 820 = 82$ ريال.}

\bigskip

\text{٥٩ إذا كان $x + 2y = 22$ ، $x = 2$}
\text{قارن بين:}
\begin{itemize}
    \item \text{القيمة الأولى: $12$}
    \item \text{القيمة الثانية: $\frac{x+y}{12}$}
\end{itemize}
\begin{itemize}
    \item[أ] \text{القيمة الأولى أكبر}
    \item[ب] \text{القيمة الثانية أكبر}
    \item[ج] \text{القيمتان متساويتان}
    \item[د] \text{المعلومات غير كافية}
\end{itemize}
\text{الحل:}
\begin{align*}
x + 2y &= 22 \\
2 + 2y &= 22 \\
2y &= 20 \\
y &= 10
\end{align*}
\text{القيمة الأولى} $= 12$
\text{القيمة الثانية} $= \frac{x+y}{12} = \frac{2+10}{12} = \frac{12}{12} = 1$
\text{إذن، القيمة الأولى أكبر.}

\bigskip

\text{٦٠ إذا كان $1, 2, 3, 4, \dots, 19$}
\text{قارن بين:}
\begin{itemize}
    \item \text{القيمة الأولى: مجموع الأعداد الفردية}
    \item \text{القيمة الثانية: مجموع الأعداد الزوجية}
\end{itemize}
\begin{itemize}
    \item[أ] \text{القيمة الأولى أكبر}
    \item[ب] \text{القيمة الثانية أكبر}
    \item[ج] \text{القيمتان متساويتان}
    \item[د] \text{المعلومات غير كافية}
\end{itemize}
\text{الحل:}
\text{الأعداد الفردية هي $1, 3, \dots, 19$. عددها $10$. مجموعها $10^2 = 100$.}
\text{الأعداد الزوجية هي $2, 4, \dots, 18$. عددها $9$. مجموعها $9 \times (9+1) = 90$.}
\text{إذن، القيمة الأولى أكبر.}

\bigskip

\text{٦١ إذا كان مع محمد ٢٥٠٠ هللة و ٧ ريال}
\begin{tabular}{|l|l|}
\hline
\text{القيمة الأولى} & \text{القيمة الثانية} \\
\hline
\text{ما مع محمد} & \text{٣٠ ريال} \\
\hline
\end{tabular}
\begin{itemize}
    \item[أ] \text{القيمة الأولى أكبر}
    \item[ب] \text{القيمة الثانية أكبر}
    \item[ج] \text{القيمتان متساويتان}
    \item[د] \text{المعطيات غير كافية}
\end{itemize}
\text{الحل:}
\text{ما مع محمد = $2500$ هللة + $7$ ريال}
\text{$2500$ هللة = $\frac{2500}{100}$ ريال = $25$ ريال.}
\text{إذن، ما مع محمد = $25 + 7 = 32$ ريال.}
\text{القيمة الأولى = $32$ ريال.}
\text{القيمة الثانية = $30$ ريال.}
\text{إذن، القيمة الأولى أكبر.}
