قاعدة ٥ الأس الزوجي والأس الفردي
* الأس الزوجي للعدد السالب يعطي ناتج موجباً
  * مثال (-٢)⁴ = ١٦
* الأس الفردي للعدد السالب يعطي ناتج سالباً
  * مثال (-٢)³ = -٨

٥٧ قارن بين
القيمة الأولى (-١/٢)²
القيمة الثانية (-١/٢)⁵
الحل
القيمة الأولى أكبر لأن الناتج موجب (أ)

٥٨ قارن بين
القيمة الأولى (-٥)⁶
القيمة الثانية (-٥)⁷
الحل
ناتج القيمة الأولى موجب، وناتج القيمة الثانية سالب
أي أن القيمة الأولى أكبر (أ)

٥٩ قارن بين
القيمة الأولى - ٥⁶
القيمة الثانية - ٥³
الحل
حيث أن قيمة ٥⁶ أكبر من ٥³ لاحظ السالب خارج الأس
فإن - ٥⁶ أصغر من - ٥³
لذلك فإن القيمة الثانية أكبر (ب)

٦٠ قارن بين
القيمة الأولى - ٢ - ٢ - ٢ - ٢ - ٢

الحل
القيمة الأولى = - ١٠
القيمة الثانية = - ٣٢
أي أن القيمة الأولى أكبر (أ)

\section*{قاعدة ٥ الأس الزوجي والأس الفردي}

\begin{itemize}

    \end{itemize}

\section*{٥٧ قارن بين}
القيمة الأولى $\left(-\frac{1}{2}\right)^2$
القيمة الثانية $\left(-\frac{1}{2}\right)^5$

\subsection*{الحل}
القيمة الأولى أكبر لأن الناتج موجب (أ)

\section*{٥٨ قارن بين}
القيمة الأولى $(-5)^6$
القيمة الثانية $(-5)^7$

\subsection*{الحل}
ناتج القيمة الأولى موجب، وناتج القيمة الثانية سالب\\
أي أن القيمة الأولى أكبر (أ)

\section*{٥٩ قارن بين}
القيمة الأولى $-5^6$
القيمة الثانية $-5^3$

\subsection*{الحل}
حيث أن قيمة $5^6$ أكبر من $5^3$ (لاحظ السالب خارج الأس)\\
فإن $-5^6 < -5^3$\\
لذلك فإن القيمة الثانية أكبر (ب)

\section*{٦٠ قارن بين}
القيمة الأولى $-2 - 2 - 2 - 2 - 2$
القيمة الثانية $(-2)^5$

\subsection*{الحل}
القيمة الأولى $= -10$\\
القيمة الثانية $= -32$\\
أي أن القيمة الأولى أكبر (أ)
