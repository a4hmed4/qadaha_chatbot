🧠 النص المستخرج:
مهارات و قوانين ما قبل التأسيس

مثال ١٤
ما قيمة
٥٨٠ + ٩٠٠ + ٢٠٠ + ١١٠ + ٥٠٠ + ٦٠٠ + ٧٠٠ + ٢٠٠ + ١١٠ + ٩٠٠
أ ٥٠٠
ب ٥٥٠
ج ٤٥٠٠
د ٦٠٠٠

الحل
نختار الاعداد التي تجمع مع بعضها بسهولة
٥٨٠ + ٩٠٠ + ٢٠٠ + ١١٠ + ٥٠٠ + ٦٠٠ + ٧٠٠ + ٢٠٠ + ١١٠ + ٩٠٠
٢٧٠٠ = ٣٠٠ + ٦٠٠ + ٩٠٠ + ٩٠٠
٨٠٠ = ٥٨٠ + ١١٠ + ١١٠
١٥٠٠ = ٥٠٠ + ٧٠٠ + ٣٠٠
المجموع = ٢٧٠٠ + ٨٠٠ + ١٥٠٠ = ٥٠٠٠

مثال ١٥
ما قيمة
٧٥٠ + ٢٥٠ + ٣٠٠ + ١٨٠ + ٨٢٠ + ٧٠٠ + ٣٥٠ + ٦٥٠ + ١٠ + ٩٩٠
أ ٥٠٠
ب ٥٥٠٠
ج ٤٥٠٠
د ٦٠٠٠

الحل
٧٥٠ + ٢٥٠ + ٣٠٠ + ١٨٠ + ٨٢٠ + ٧٠٠ + ٣٥٠ + ٦٥٠ + ١٠ + ٩٩٠
١٠٠٠ = ٣٥٠ + ٦٥٠
١٠٠٠ = ١٠ + ٩٩٠
١٠٠٠ = ٣٠٠ + ٧٠٠
١٠٠٠ = ١٨٠ + ٨٢٠
١٠٠٠ = ٧٥٠ + ٢٥٠
المجموع = ١٠٠٠ + ١٠٠٠ + ١٠٠٠ + ١٠٠٠ + ١٠٠٠ = ٥٠٠٠

٥ ضرب الاعداد الكبيرة
• في حالة الضرب نعتمد على ضرب الأحاد فقط
حيث ضرب الأحاد في كل عدد يعطي أحاد الناتج
• خانة العشرات للناتج = الاحاد الناتج من العملية التالية
= احاد الأول × عشرات الثاني + احاد الثاني × عشرات الاول

مثال ١٦
أوجد ناتج ٦٣ × ٢٧٤ × ٤
أ ٦٩٠٤٨
ب ٩٧٨٤٢
ج ٧٢٥٦٩
د ٦٩٨٤٠

الحل
نحاول ضرب آحاد كل عدد ٣ × ٤ × ٤ = ٤٨ نجد أن آحاد الناتج هو ٨
لذلك نختار العدد الذي آحاده ٨ وهو ٦٩٠٤٨
