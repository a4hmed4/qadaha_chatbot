\textbf{مثال 39}
تبسيط المقدار $(s + p)^2 - 2sp + (s - p)^2$
أ $s^2 + p^2$
ب $2s^2 + 2p^2$
ج $2sp$
د $4sp$

\textbf{الحل}
\text{بفك الأقواس}
$$s^2 + 2sp + p^2 - 2sp + s^2 - 2sp + p^2 =$$
$$s^2 + p^2 + s^2 + p^2 - 2sp = 2s^2 + 2p^2 \quad (\text{ب})$$

\textbf{مثال 40}
\text{إذا كان } $s = 3\sqrt{2} \text{، } p = 2\sqrt{2} \text{ اوجد } s^2 - 2sp + p^2$
أ $2$
ب $6$
ج $12$
د $2\sqrt{2}$

\textbf{الحل}
\text{حيث ان } $s^2 - 2sp + p^2 = (s - p)^2$
$$= (3\sqrt{2} - 2\sqrt{2})^2$$
$$= (\sqrt{2})^2$$
$$= 2$$

\textbf{خالي بالك}
$(s + p)^2 > s^2 + p^2$
\text{حيث } $s, p \text{ اعداد موجبة}$

\textbf{مثال 41}
\textbf{قارن بين}
\text{القيمة الأولى } $49$
\text{القيمة الثانية } $12^2 + 37^2$
أ \text{القيمة الأولى اكبر}
ب \text{القيمة الثانية اكبر}
ج \text{القيمتان متساويتان}

\textbf{الحل}
\text{حيث أن } $49 = 12 + 37 \text{ فإن}$
\text{القيمة الأولى } $(12 + 37)^2$
\text{ومن الملاحظة الأولى تكون القيمة الأولى اكبر } (\text{أ})
