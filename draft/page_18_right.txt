مهارات و قوانين ما قبل التأسيس

مثال ٦١
المثلثان متشابهان
اوجد قيمة س
أ ٦
ب ٧
ج ٨
د ٩
الحل
حيث ان المثلثات متشابهة فإن
$\frac{١+س}{٢-س} = \frac{٨}{١٢}$ أي ان $\frac{١+س}{٢-س} = \frac{٢}{٣}$
$٤س - ٦ = ٣س + ٣$
$س = ٩$

مثال ٦٢
ما قيمة س في الشكل
أ ٦
ب ٩
ج ٣
د ١٢
الحل
حيث ان المثلث الصغير و الكبير متشابهة فإن
$\frac{س}{١٥} = \frac{٤}{١٠}$
$س = \frac{١٥ \times ٤}{١٠}$
$س = ٦$

مثال ٦٣
في الشكل المجاور
اوجد قيمة س
أ ٤,١
ب ٦,١
ج ٨,١
د ٢
الحل
المثلث الكبير يشابه المثلث الصغير
$\frac{٩}{س} = \frac{١٠}{٢}$
$س = \frac{٩ \times ٢}{١٠}$
$س = \frac{١٨}{١٠}$
$س = ١,٨$

\paragraph{مثال 62}
ما قيمة $x$ في الشكل\\
أ 6\\
ب 9\\
ج 3\\
د 12

\textbf{الحل}\\
حيث ان المثلث الصغير و الكبير متشابهة فإن
$$ \frac{x}{15} = \frac{4}{10} $$
$$ x = \frac{15 \times 4}{10} $$
$$ x = 6 $$

\paragraph{مثال 63}
في الشكل المجاور\\
اوجد قيمة $x$\\
أ 4.1\\
ب 6.1\\
ج 8.1\\
د 2

\textbf{الحل}\\
المثلث الكبير يشابه المثلث الصغير
$$ \frac{9}{x} = \frac{10}{2} $$
$$ x = \frac{9 \times 2}{10} $$
$$ x = \frac{18}{10} $$
$$ x = 1.8 $$
