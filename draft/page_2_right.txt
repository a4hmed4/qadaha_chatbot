```latex
% يجب تجميع هذا المستند باستخدام XeLaTeX أو LuaLaTeX للحصول على أفضل دعم للغة العربية.
\documentclass{article}

% دعم اللغة العربية
\usepackage{polyglossia}
\setdefaultlanguage[locale=default]{arabic}
\setotherlanguage{english} % لدعم الروابط الإنجليزية
\newfontfamily\arabicfont[Script=Arabic]{Amiri} % يمكنك استبدال 'Amiri' بأي خط عربي آخر مثل Noto Sans Arabic, Scheherazade, إلخ.

% لدعم الروابط
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=blue,
    urlcolor=blue,
}

% لدعم علامة الصح (checkmark)
\usepackage{amsfonts} % توفر \checkmark

% لإعداد الهوامش والتنسيق العام
\usepackage{geometry}
\geometry{a4paper, margin=1in}
\usepackage{amsmath} % مفيد للرياضيات، على الرغم من عدم استخدامه مباشرة هنا

\begin{document}

% عنوان الجزء الأول
\begin{center}
    \textbf{دورات المعاصر} \\
    \url{iqdr.com} \\

\end{center}

\vspace{1cm} % مسافة رأسية

    \item[\checkmark] شرح اساسيات التحصيلي رياضيات - فيزياء - كيمياء

    \item[\checkmark] قسم مجاني للتدريب و قياس المستوى
    \item[\checkmark] قسم مجاني خاص بنماذج المختبرين اليومية مع الشرح
```
