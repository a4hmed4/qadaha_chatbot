\textbf{مثال ٩-١} \\
ما العدد الذي إذا قسمته على $3, 4, 5$ يكون الباقي $2$
\begin{itemize}
    \item[أ] $52$
    \item[ب] $42$
    \item[ج] $62$
    \item[د] $72$
\end{itemize}

\textbf{الحل} \\
نبدأ الحل من الخيارات بأن نبحث عن العدد الذي يحقق التمرين
\begin{itemize}
    \item[أ] $52 \leftarrow$ نقسم $52$ على $3$ يكون الناتج $17$ والباقي $1$ \\
    ويكون الحل خطأ
    \item[ب] $42 \leftarrow$ نقسم $42$ على $3$ يكون الناتج $14$ والباقي صفر \\
    أي أن الحل خطأ
    \item[ج] $62 \leftarrow$ نقسم $62$ على $3$ يكون الناتج $20$ والباقي $2$ \\
    نقسم $62$ على $4$ يكون الناتج $15$ والباقي $2$ \\
    نقسم $62$ على $5$ يكون الناتج $12$ والباقي $2$ \\
    وبذلك يكون جـ $62$ هو الحل الصحيح
\end{itemize}

\vspace{0.5cm} % Add some vertical space

\textbf{مثال ١٠-١} \\
إذا كان $أ$ عدد فردي فإن باقي قسمة $أ^2$ على $4$ هو
\begin{itemize}
    \item[أ] $1$
    \item[ب] $2$
    \item[ج] $3$
    \item[د] صفر
\end{itemize}

\textbf{الحل} \\
نختار $أ$ بأي عدد فردي مثلاً $أ = 3$ فيكون $أ^2 = 9$ \\
نقسم $9$ على $4$ يكون الباقي $1$ أي أن الحل أ $1$

\vspace{0.5cm}

\textbf{مثال ١١-١} \\
إذا كان $\frac{9}{س} = ص$ والباقي $2$ أوجد $س$
\begin{itemize}
    \item[أ] $9$
    \item[ب] $5$
    \item[ج] $6$
    \item[د] $7$
\end{itemize}

\textbf{الحل} \\
نبدأ من الخيارات ونبحث عن العدد الذي إذا قسمت عليه $9$ يكون الباقي $2$ \\
نجد أن العدد هو د $7$

\vspace{0.5cm}

\textbf{مثال ١٢-١} \\
ما باقي قسمة $101$ على $3$ ؟
\begin{itemize}
    \item[أ] $1$
    \item[ب] $2$
    \item[ج] $3$
    \item[د] $4$
\end{itemize}

\textbf{الحل} \\
عند قسمة $101$ على $3$ يكون الناتج $33$ والباقي هو $2$
