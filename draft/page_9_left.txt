٣ الجمع بالعامل المشترك
يمكن جمع عدد كبير من الاعداد بأخذ عامل مشترك منهم

مثال ١١
ما ناتج جمع ٤٤ + ٥٥ + ٦٦ + ٧٧ + ٨٨ + ٩٩
أ ٥١٢
ب ٩١٣
ج ٧١٧
د ٤٢٩
الحل
بأخذ ١١ عامل مشترك
١١ ( ٤ + ٥ + ٦ + ٧ + ٨ + ٩ ) = ١١ × ٣٩ = ٤٢٩ ( د )

مثال ١٢
قارن بين
القيمة الأولى ٧٧+٦٦+٩٩+٥٥ / ٥٥+٤٤
القيمة الثانية ٣
أ القيمة الأولى أكبر
ب القيمة الثانية أكبر
ج القيمتان متساويتان
د المعلومات غير كافية
الحل
بأخذ العامل المشترك بسطا و مقاما في القيمة الأولى
القيمة الأولى = ( ٧ + ٦ + ٩ + ٥ ) × ١١ / ( ٥ + ٤ ) × ١١ = ٢٧ × ١١ / ٩ × ١١ = ٣
معنى ذلك ان القيمتان متساويتان ( ج )

٤ جمع و طرح الاعداد الكبيرة
نعتمد على تجميع الأرقام التي يمكن جمعها أو طرحها مع بعض بسهولة لتعطي أعداداً أولها أصفار

مثال ١٣
أوجد ناتج جمع ٦٥ + ٩٤ + ١٣٢ + ٣٥ + ٦٨ + ٦
أ ٣٥٠
ب ٤٠٠
ج ٤٢٠
د ٥٠٠
الحل
نختار الأعداد التي تجمع مع بعضها بسهولة
٦٥ + ٣٥ = ١٠٠
٩٤ + ٦ = ١٠٠
١٣٢ + ٦٨ = ٢٠٠
وبذلك يصبح الناتج بسهولة
١٠٠ + ١٠٠ + ٢٠٠ = ٤٠٠ ( ب )

\section*{٣. الجمع بالعامل المشترك}
يمكن جمع عدد كبير من الأعداد بأخذ عامل مشترك منهم.

\subsection*{مثال ١١}
ما ناتج جمع $44 + 55 + 66 + 77 + 88 + 99$؟\\
أ) $512$\\
ب) $913$\\
ج) $717$\\
د) $429$\\

\textbf{الحل}\\
بأخذ $11$ عامل مشترك:\\
$11 ( 4 + 5 + 6 + 7 + 8 + 9 ) = 11 \times 39 = 429$ (د)

\subsection*{مثال ١٢}
قارن بين:\\
القيمة الأولى $\frac{77+66+99+55}{55+44}$\\
القيمة الثانية $3$\\
أ) القيمة الأولى أكبر\\
ب) القيمة الثانية أكبر\\
ج) القيمتان متساويتان\\
د) المعلومات غير كافية\\

\textbf{الحل}\\
بأخذ العامل المشترك بسطًا ومقامًا في القيمة الأولى:\\
القيمة الأولى $= \frac{(7 + 6 + 9 + 5) \times 11}{(5 + 4) \times 11} = \frac{27 \times 11}{9 \times 11} = 3$\\
معنى ذلك أن القيمتين متساويتان (ج)

\section*{٤. جمع وطرح الأعداد الكبيرة}
نعتمد على تجميع الأرقام التي يمكن جمعها أو طرحها مع بعض بسهولة لتعطي أعدادًا أولها أصفار.

\subsection*{مثال ١٣}
أوجد ناتج جمع $65 + 94 + 132 + 35 + 68 + 6$\\
أ) $350$\\
ب) $400$\\
ج) $420$\\
د) $500$\\

\textbf{الحل}\\
نختار الأعداد التي تُجمع مع بعضها بسهولة:\\
$65 + 35 = 100$\\
$94 + 6 = 100$\\
$132 + 68 = 200$\\
وبذلك يصبح الناتج بسهولة:\\
$100 + 100 + 200 = 400$ (ب)
