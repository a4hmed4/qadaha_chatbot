مهارات و قوانين ما قبل الـ
٦ حساب عدد المربعات
نرقم المربعات الافقية ثم نربع كل عدد و نجمعهم

مثال ٢١
ما عدد المربعات في الشكل
أ ١٢
ب ١٣
ج ١٤
د ١٥
الحل
نرقم المربعات افقيا ١, ٢, ٣
نربع الترقيم ١, ٤, ٩
نجمع المربعات ١ + ٤ + ٩ = ١٤ مربع

مثال ٢٢
ما عدد المربعات في الشكل
أ ١١
ب ١٢
ج ١٣
د ١٤
الحل
نعد المربعات الصغيرة نجدها ٨
نعد المربعات الكبيرة نجدها
الأزرق و الأحمر و الأخضر عددهم ٣
يكون عدد المربعات كاملة = ٨ + ٣ = ١١

٧ حساب عدد المستطيلات
نرقم المستطيلات افقي و رأسي
عدد المستطيلات = مجموع الافقي × مجموع الرأسي

مثال ٢٣
كم عدد المستطيلات في الشكل
أ ١٢
ب ١٤
ج ١٦
د ١٨
الحل
نرقم افقي يكون ١, ٢, ٣, يكون المجموع = ٦
نرقم الرأسي يكون ١, ٢, يكون المجموع = ٣
عدد المستطيلات = ٣ × ٦ = ١٨

\subsection*{٦ حساب عدد المربعات}
نرقم المربعات الأفقية ثم نربع كل عدد ونجمعهم.

\subsubsection*{مثال ٢١}
ما عدد المربعات في الشكل؟

\begin{enumerate}[label=\alph*]
    \item ١٢
    \item ١٣
    \item ١٤
    \item ١٥
\end{enumerate}

\paragraph*{الحل:}
نرقم المربعات أفقيًا: ١، ٢، ٣.

نربع الترقيم:
\(1^2 = 1\)
\(2^2 = 4\)
\(3^2 = 9\)

نجمع المربعات: \(1 + 4 + 9 = 14 \text{ مربع}\)

\subsubsection*{مثال ٢٢}
ما عدد المربعات في الشكل؟

\begin{enumerate}[label=\alph*]
    \item ١١
    \item ١٢
    \item ١٣
    \item ١٤
\end{enumerate}

\paragraph*{الحل:}
نعد المربعات الصغيرة نجدها ٨.
نعد المربعات الكبيرة نجدها:
الأزرق والأحمر والأخضر عددهم ٣.
يكون عدد المربعات كاملة = \(8 + 3 = 11\)

\subsection*{٧ حساب عدد المستطيلات}
نرقم المستطيلات أفقيًا ورأسيًا.
عدد المستطيلات = مجموع الأفقي \(\times\) مجموع الرأسي

\subsubsection*{مثال ٢٣}
كم عدد المستطيلات في الشكل؟

\begin{enumerate}[label=\alph*]
    \item ١٢
    \item ١٤
    \item ١٦
    \item ١٨
\end{enumerate}

\paragraph*{الحل:}
نرقم أفقيًا يكون ١، ٢، ٣، يكون المجموع = \(6\).
نرقم رأسيًا يكون ١، ٢، يكون المجموع = \(3\).
عدد المستطيلات = \(3 \times 6 = 18\)
