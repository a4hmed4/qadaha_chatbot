\section*{المسائل الرياضية}

\subsection*{١٦. مقارنة قيم ص}
إذا كانت $ ص = س^٢ - ١ $ قارن بين:
\begin{itemize}
    \item القيمة الأولى: قيمة $ ص $ عندما $ س = ٢ $
    \item القيمة الثانية: قيمة $ ص $ عندما $ س = -٢ $
\end{itemize}

\textbf{الحل:}
\begin{itemize}
    \item القيمة الأولى: عندما $ س = ٢ $ يصبح المقدار $ = (٢)^٢ - ١ = ٤ - ١ = ٣ $
    \item القيمة الثانية: عندما $ س = -٢ $ يصبح المقدار $ = (-٢)^٢ - ١ = ٤ - ١ = ٣ $
\end{itemize}
إذن القيمتان متساويتان (ج)

\bigskip

\subsection*{١٦. مقارنة ك وع}
إذا كان $ ٣ ك = ٨١ $، و $ ٤٠ \times ع = ٨٠ $ قارن بين:
\begin{itemize}
    \item القيمة الأولى: $ ك $
    \item القيمة الثانية: $ ع $
\end{itemize}

\textbf{الحل:}
\begin{itemize}
    \item $ ٣ ك = ٨١ \implies ك = ٤ $ (ملاحظة: حسب النص الأصلي، رغم أن $٨١/٣ = ٢٧$)
    \item $ ٤٠ \times ع = ٨٠ \implies ع = ٢ $
\end{itemize}
وبذلك تكون القيمة الأولى أكبر (أ)

\bigskip

\subsection*{١٦. إيجاد ن}
إذا كان $ ٩٩٩ \times ١١١ = ٣ \times ٣ \times ن^٢ $ أوجد $ ن $
\begin{itemize}
    \item أ) $ ٣ $
    \item ب) $ ٣٣٣ $
    \item ج) $ ١١١ $
    \item د) $ ١ $
\end{itemize}

\textbf{الحل:}
$ ٩٩٩ \times ١١١ = ٩ \times ن^٢ $
نقسم على $ ٩ $:
$ \frac{٩٩٩}{٩} \times ١١١ = ن^٢ $
$ ١١١ \times ١١١ = ن^٢ $
أي أن $ ن = ١١١ $ (ج)

\bigskip

\subsection*{١٧. قيمة س}
إذا كان $ س \times س \times س = ٨١ $ فما قيمة $ س $
\begin{itemize}
    \item أ) $ ٣ $
    \item ب) $ ٤ $
    \item ج) $ ٩ $
    \item د) $ ٢٧ $
\end{itemize}

\textbf{الحل:}
بتجربة الخيارات:
نجد أن العدد الذي يحقق المعادلة هو $ س = ٣ $ (أ)
(ملاحظة: $ ٣^٣ = ٢٧ $، وليس $ ٨١ $. يتم تحويل النص كما هو.)

\bigskip

\subsection*{١٨. قيمة ك}
إذا كان $ ٩ ك - ٨ ك = ٠ $ فإن قيمة $ ك = $
\begin{itemize}
    \item أ) $ ٠ $
    \item ب) $ ١ $
    \item ج) $ ٢ $
    \item د) $ -١ $
\end{itemize}

\textbf{الحل:}
بتجربة الخيارات نبحث عن قيمة $ ك $ التي تحقق المعادلة نجد:
أنها $ ك = ٠ $ (أ)

\bigskip

\subsection*{١٩. مقارنة ك و ١}
إذا كان $ ٩ ك - ٨ ك = ١ $ قارن بين:
\begin{itemize}
    \item القيمة الأولى: $ ك $
    \item القيمة الثانية: $ ١ $
\end{itemize}

\end{document}
